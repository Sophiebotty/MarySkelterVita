cmake_minimum_required(VERSION 3.18)   # find_library/find_path ... REQUIRED needs 3.18

# ---------------------------------------------------------------------------
# Vita ResProto - standalone resource-loading prototype (Phase 8).
#
# Separate project, separate build directory, separate title ID: building or
# installing it never touches the main app (VDNG00001) or VitaBench
# (VDNG00002). Shared UI/input modules from ../src are compiled unmodified.
#
# It reads, validates and decompresses ONLY data it generates itself or
# fixtures produced by tools/make_resproto_fixtures.py. It does not read any
# Mary Skelter game file and cannot load Finale archives.
# ---------------------------------------------------------------------------
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(DEFINED ENV{VITASDK})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "VitaSDK toolchain file")
  elseif(EXISTS "/usr/local/vitasdk/share/vita.toolchain.cmake")
    set(CMAKE_TOOLCHAIN_FILE "/usr/local/vitasdk/share/vita.toolchain.cmake" CACHE PATH "VitaSDK toolchain file")
  else()
    message(FATAL_ERROR "VitaSDK not found. Set VITASDK or pass -DCMAKE_TOOLCHAIN_FILE.")
  endif()
endif()

project(VitaResProto C CXX)   # CXX only for linking libvitaGL's C++ objects

include("${VITASDK}/share/vita.cmake" REQUIRED)

set(VITA_APP_NAME "Vita ResProto")
set(VITA_TITLEID  "VDNG00003")
set(VITA_VERSION  "01.00")

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

find_package(SDL2 REQUIRED CONFIG)

get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)

# zlib: VitaSDK package "zlib" (installed with vdpm; libz.a + zlib.h).
find_library(ZLIB_LIB z REQUIRED)
find_path(ZLIB_INC zlib.h REQUIRED)

add_executable(${PROJECT_NAME}
  src/rp_main.c
  src/rp_tests.c
  src/rp_fixture.c
  src/rp_container.c
  src/rp_inflate.c
  src/rp_io.c
  src/rp_mem.c
  src/rp_report.c
  src/rp_time.c
  # shared modules (unmodified)
  ${PROJECT_ROOT}/src/graphics/hud2d.c
  ${PROJECT_ROOT}/src/input/input.c
)

target_include_directories(${PROJECT_NAME} PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/src
  ${PROJECT_ROOT}/src
  ${ZLIB_INC}
)

target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra)

target_link_libraries(${PROJECT_NAME} PRIVATE
  SDL2::SDL2-static
  ${ZLIB_LIB}
  SceSysmem_stub      # sceKernelGetFreeMemorySize (memory snapshots)
)
set_target_properties(${PROJECT_NAME} PROPERTIES LINKER_LANGUAGE CXX)

vita_create_self(eboot.bin ${PROJECT_NAME})

# LiveArea art reuses the project's original placeholder images.
# fixtures/ (generated by tools/make_resproto_fixtures.py) is installed with
# the app and read back from ux0:app/VDNG00003/fixtures/.
vita_create_vpk(${PROJECT_NAME}.vpk ${VITA_TITLEID} eboot.bin
  VERSION ${VITA_VERSION}
  NAME    ${VITA_APP_NAME}
  FILE    ${PROJECT_ROOT}/sce_sys/icon0.png                        sce_sys/icon0.png
  FILE    ${PROJECT_ROOT}/sce_sys/livearea/contents/bg.png         sce_sys/livearea/contents/bg.png
  FILE    ${PROJECT_ROOT}/sce_sys/livearea/contents/startup.png    sce_sys/livearea/contents/startup.png
  FILE    ${PROJECT_ROOT}/sce_sys/livearea/contents/template.xml   sce_sys/livearea/contents/template.xml
  FILE    ${CMAKE_CURRENT_SOURCE_DIR}/fixtures                     fixtures
)
